% Preamble additions for the latexdiff output.
% PNAS's \matmethods{#1} stores its argument in \@matmethods and is later
% displayed by \showmatmethods. When latexdiff emits BOTH an added and a
% deleted \matmethods{...} block, the second call overwrites the first, so
% one side of the diff is lost. This redefinition makes subsequent calls
% append rather than overwrite, so added and deleted methods content both
% render in the compiled diff.
\makeatletter
\let\origmatmethods\matmethods
\renewcommand{\matmethods}[1]{%
  \@ifundefined{@matmethods}%
    {\origmatmethods{#1}}%
    {\g@addto@macro\@matmethods{#1}}%
}
\makeatother
